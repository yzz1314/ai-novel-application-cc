```java
package com.novel.system.service;

    public Project createProject(String name, String description,

            "prompts",

        for (String dir : directories) {

        }
        
        // 创建记忆文件
        Files.writeString(projectRoot.resolve("memory/characters.md"), "# 人物状态记忆\n\n");
```

```java
package com.novel.system.service;

import com.novel.system.entity.Sample;
import com.novel.system.entity.SampleStatus;
import com.novel.system.repository.SampleRepository;
import com.novel.system.util.FileUtils;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.web.multipart.MultipartFile;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;

@Slf4j
@Service
@RequiredArgsConstructor
public class FileStorageService {
    
    private final SampleRepository sampleRepository;
    private final String projectBasePath = "projects/";
    
    public Sample uploadSample(String projectId, MultipartFile file) 
            throws IOException, NoSuchAlgorithmException {
        
        log.info("Uploading sample for project: {}", projectId);
        
        // 1. 验证文件
        validateFile(file);
        
        // 2. 计算文件哈希
        String fileHash = calculateFileHash(file.getBytes());
        
        // 3. 保存原始文件
        String fileName = file.getOriginalFilename();
        Path rawFilePath = Paths.get(projectBasePath, projectId, "samples/raw", fileName);
        Files.write(rawFilePath, file.getBytes());
        
        // 4. 创建Sample记录
        Sample sample = new Sample();
        sample.setProjectId(projectId);
        sample.setFileName(fileName);
        sample.setFilePath(rawFilePath.toString());
        sample.setFileHash(fileHash);
        sample.setFileSizeBytes(file.getSize());
        sample.setTotalChars(0); // 待文本规范化后更新
        sample.setStatus(SampleStatus.UPLOADED);
        
        sample = sampleRepository.save(sample);
        
        log.info("Sample uploaded with ID: {}", sample.getId());
        
        return sample;
    }
    
    private void validateFile(MultipartFile file) {
        if (file.isEmpty()) {
            throw new IllegalArgumentException("File is empty");
        }
        
        String fileName = file.getOriginalFilename();
        if (!fileName.endsWith(".md") && !fileName.endsWith(".txt")) {
            throw new IllegalArgumentException("Only .md and .txt files are supported");
        }
        
        // 最大20MB
        if (file.getSize() > 20 * 1024 * 1024) {
            throw new IllegalArgumentException("File size exceeds 20MB");
        }
    }
    
    private String calculateFileHash(byte[] content) throws NoSuchAlgorithmException {
        MessageDigest md = MessageDigest.getInstance("SHA-256");
        byte[] hash = md.digest(content);
        StringBuilder hexString = new StringBuilder();
        for (byte b : hash) {
            String hex = Integer.toHexString(0xff & b);
            if (hex.length() == 1) hexString.append('0');
            hexString.append(hex);
        }
        return hexString.toString();
    }
    
    public String readSampleContent(String sampleId) throws IOException {
        Sample sample = sampleRepository.findById(sampleId)
            .orElseThrow(() -> new IllegalArgumentException("Sample not found"));
        
        Path filePath = Paths.get(sample.getFilePath());
        return Files.readString(filePath);
    }
}
```

#### 2.4.3 TaskExecutorService

```java
package com.novel.system.service;

import com.novel.system.entity.Task;
import com.novel.system.entity.TaskStatus;
import com.novel.system.repository.TaskRepository;
import com.novel.system.dto.AgentRequest;
import com.novel.system.dto.AgentResponse;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.time.LocalDateTime;
import java.util.Map;
import java.util.concurrent.CompletableFuture;

@Slf4j
@Service
@RequiredArgsConstructor
public class TaskExecutorService {
    
    private final TaskRepository taskRepository;
    private final PythonClientService pythonClientService;
    
    @Transactional
    public Task createTask(String projectId, String taskType, String agentName,
                          Map<String, Object> inputRefs, Map<String, Object> parameters) {
        
        Task task = new Task();
        task.setProjectId(projectId);
        task.setTaskType(taskType);
        task.setAgentName(agentName);
        task.setInputRefs(inputRefs);
        task.setParameters(parameters);
        task.setStatus(TaskStatus.PENDING);
        
        return taskRepository.save(task);
    }
    
    @Async
    @Transactional
    public CompletableFuture<Task> executeTaskAsync(String taskId) {
        log.info("Starting async execution of task: {}", taskId);
        
        Task task = taskRepository.findById(taskId)
            .orElseThrow(() -> new IllegalArgumentException("Task not found"));
        
        try {
            // 更新状态为运行中
            task.setStatus(TaskStatus.RUNNING);
            task.setStartedAt(LocalDateTime.now());
            taskRepository.save(task);
            
            // 构建AgentRequest
            AgentRequest request = AgentRequest.builder()
                .taskId(task.getId())
                .projectId(task.getProjectId())
                .taskType(task.getTaskType())
                .inputRefs(task.getInputRefs())
                .parameters(task.getParameters())
                .build();
            
            // 调用Python服务
            AgentResponse response = pythonClientService.callAgent(task.getAgentName(), request);
            
            // 更新任务结果
            task.setStatus(TaskStatus.valueOf(response.getStatus()));
            task.setOutputRefs(response.getOutputRefs());
            task.setResult(response.getStructuredOutput());
            task.setErrors(response.getErrors());
            task.setWarnings(response.getWarnings());
            task.setMetrics(response.getMetrics());
            task.setCheckpointRef(response.getCheckpointRef());
            task.setFinishedAt(LocalDateTime.now());
            
            taskRepository.save(task);
            
            log.info("Task completed successfully: {}", taskId);
            
        } catch (Exception e) {
            log.error("Task failed: {}", taskId, e);
            
            task.setStatus(TaskStatus.FAILED);
            task.setErrors(Map.of(
                "error_type", e.getClass().getSimpleName(),
                "error_message", e.getMessage()
            ));
            task.setFinishedAt(LocalDateTime.now());
            
            taskRepository.save(task);
            
            // 检查是否需要重试
            if (task.getRetryCount() < task.getMaxRetries() && isRetryable(e)) {
                task.setRetryCount(task.getRetryCount() + 1);
                task.setStatus(TaskStatus.PENDING);
                taskRepository.save(task);
                
                log.info("Retrying task: {} (attempt {})", taskId, task.getRetryCount());
                return executeTaskAsync(taskId);
            }
        }
        
        return CompletableFuture.completedFuture(task);
    }
    
    private boolean isRetryable(Exception e) {
        // 网络错误、超时、速率限制等可重试
        return e instanceof java.net.SocketTimeoutException ||
               e instanceof java.net.ConnectException ||
               e.getMessage().contains("rate limit");
    }
    
    public Task getTask(String taskId) {
        return taskRepository.findById(taskId)
            .orElseThrow(() -> new IllegalArgumentException("Task not found"));
    }
    
    @Transactional
    public Task cancelTask(String taskId) {
        Task task = getTask(taskId);
        
        if (task.getStatus() == TaskStatus.RUNNING) {
            // 通知Python服务取消任务
            pythonClientService.cancelTask(taskId);
        }
        
        task.setStatus(TaskStatus.CANCELLED);
        task.setFinishedAt(LocalDateTime.now());
        
        return taskRepository.save(task);
    }
}
```

```java
package com.novel.system.service;

import com.novel.system.dto.AgentRequest;
import com.novel.system.dto.AgentResponse;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.*;
import org.springframework.stereotype.Service;
import org.springframework.web.client.RestTemplate;

@Slf4j
@Service
@RequiredArgsConstructor
public class PythonClientService {
    
    private final RestTemplate restTemplate;
    
    @Value("${python.service.url:http://localhost:8000}")
    private String pythonServiceUrl;
    
    public AgentResponse callAgent(String agentName, AgentRequest request) {
        String url = pythonServiceUrl + "/api/agents/" + agentName + "/run";
        
        log.info("Calling Python agent: {} at {}", agentName, url);
        
        HttpHeaders headers = new HttpHeaders();
        headers.setContentType(MediaType.APPLICATION_JSON);
        
        HttpEntity<AgentRequest> entity = new HttpEntity<>(request, headers);
        
        ResponseEntity<AgentResponse> response = restTemplate.exchange(
            url,
```

### 3.3 BaseAgent实现

```python
# agents/base.py
from abc import ABC, abstractmethod
from typing import Dict, Any, List, Optional
from datetime import datetime
import asyncio
import json
from pathlib import Path

from schemas.agent_request import AgentRequest
from schemas.agent_response import AgentResponse, MetricsInfo
from llm.client import LLMClient
from skills.skill_loader import SkillLoader
from utils.logger import get_logger

class BaseAgent(ABC):
    def __init__(self, agent_name: str):
        self.agent_name = agent_name
        self.supported_tasks: List[str] = []
        self.llm_client = LLMClient()
        self.skill_loader = SkillLoader()
        self.logger = get_logger(agent_name)
        
        # 指标记录
        self.metrics = {
            "llm_calls": 0,
            "input_tokens": 0,
            "output_tokens": 0,
            "start_time": None
        }
    
    @abstractmethod
    async def run(self, request: AgentRequest) -> AgentResponse:
        """执行Agent任务的主入口"""
        pass
    
    async def validate_request(self, request: AgentRequest) -> bool:
        """验证请求参数"""
        if not request.task_id:
            raise ValueError("task_id is required")
        if not request.project_id:
            raise ValueError("project_id is required")
        return True
    
    async def load_skills(self, skill_ids: List[str]) -> Dict[str, Any]:
        """加载Skills"""
        skills = {}
        for skill_id in skill_ids:
            skill_content = await self.skill_loader.load_skill(skill_id)
            skills[skill_id] = skill_content
        return skills
    
    async def build_context(self, request: AgentRequest) -> Dict[str, Any]:
        """构建上下文（由子类重写）"""
        return {}
    
    async def save_checkpoint(self, task_id: str, project_id: str, 
                            state: Dict[str, Any]) -> str:
        """保存checkpoint"""
        checkpoint_id = f"checkpoint_{task_id}_{int(datetime.now().timestamp())}"
        checkpoint_path = Path(f"projects/{project_id}/checkpoints/{checkpoint_id}.json")
        
        checkpoint_path.parent.mkdir(parents=True, exist_ok=True)
        
        checkpoint_data = {
            "checkpoint_id": checkpoint_id,
            "task_id": task_id,
            "timestamp": datetime.now().isoformat(),
            "state": state
        }
        
        with open(checkpoint_path, 'w', encoding='utf-8') as f:
            json.dump(checkpoint_data, f, ensure_ascii=False, indent=2)
        
        self.logger.info(f"Checkpoint saved: {checkpoint_id}")
        return checkpoint_id
    
    async def load_checkpoint(self, checkpoint_ref: str) -> Dict[str, Any]:
        """加载checkpoint"""
        checkpoint_path = Path(checkpoint_ref)
        
        if not checkpoint_path.exists():
            raise FileNotFoundError(f"Checkpoint not found: {checkpoint_ref}")
        
        with open(checkpoint_path, 'r', encoding='utf-8') as f:
            checkpoint_data = json.load(f)
        
        return checkpoint_data["state"]
    
    def _record_llm_call(self, input_tokens: int, output_tokens: int):
        """记录LLM调用指标"""
        self.metrics["llm_calls"] += 1
        self.metrics["input_tokens"] += input_tokens
        self.metrics["output_tokens"] += output_tokens
    
    def _build_response(self, request: AgentRequest, status: str,
                       output_refs: List[str] = None,
                       structured_output: Dict[str, Any] = None,
                       errors: List = None,
                       warnings: List = None,
                       checkpoint_ref: str = None) -> AgentResponse:
        """构建统一的响应对象"""
        
        duration_ms = 0
        if self.metrics["start_time"]:
            duration_ms = int((datetime.now() - self.metrics["start_time"]).total_seconds() * 1000)
        
        return AgentResponse(
            task_id=request.task_id,
            agent_name=self.agent_name,
            project_id=request.project_id,
            status=status,
            output_refs=output_refs or [],
            structured_output=structured_output or {},
            errors=errors or [],
            warnings=warnings or [],
            metrics=MetricsInfo(
                llm_calls=self.metrics["llm_calls"],
                input_tokens=self.metrics["input_tokens"],
                output_tokens=self.metrics["output_tokens"],
                duration_ms=duration_ms
            ),
            checkpoint_ref=checkpoint_ref,
            created_at=self.metrics["start_time"],
            finished_at=datetime.now()
        )
```

```python
# agents/full_text_analysis_agent.py
import asyncio
from typing import Dict, Any, List
from pathlib import Path
import json

from agents.base import BaseAgent
from schemas.agent_request import AgentRequest
from schemas.agent_response import AgentResponse
from text_processing.chunker import Chunker
from llm.prompt_builder import PromptBuilder

class FullTextAnalysisAgent(BaseAgent):
    def __init__(self):
        super().__init__("FullTextAnalysisAgent")
        self.supported_tasks = ["full_text_analysis"]
        self.chunker = Chunker()
        self.prompt_builder = PromptBuilder()
    
    async def run(self, request: AgentRequest) -> AgentResponse:
        """
        执行全文拆解任务
        """
        self.metrics["start_time"] = datetime.now()
        
        try:
            # 1. 验证请求
            await self.validate_request(request)
            
            # 2. 加载Skills
            skills = await self.load_skills(request.skill_ids or ["sample_deconstruction"])
            
            # 3. 获取样本和分块列表
            project_id = request.project_id
            sample_id = request.input_refs.get("sample_id")
            
            chunks = await self._load_chunks(project_id, sample_id)
            
            # 4. 逐块分析
            analysis_results = []
            for i, chunk in enumerate(chunks):
                self.logger.info(f"Analyzing chunk {i+1}/{len(chunks)}")
                
                # 分析单个分块
                result = await self._analyze_chunk(
                    project_id=project_id,
                    chunk=chunk,
                    skills=skills
                )
                
                analysis_results.append(result)
                
                # 定期保存checkpoint
                if (i + 1) % 10 == 0:
                    checkpoint_ref = await self.save_checkpoint(
                        request.task_id,
                        project_id,
                        {
                            "processed_chunks": i + 1,
                            "total_chunks": len(chunks),
                            "results": analysis_results
                        }
                    )
            
            # 5. 保存分析结果
            output_refs = await self._save_results(project_id, sample_id, analysis_results)
            
            # 6. 返回响应
            return self._build_response(
                request=request,
                status="success",
                output_refs=output_refs,
                structured_output={
                    "chunks_processed": len(chunks),
                    "total_techniques_extracted": sum(
                        len(r.get("prose_techniques", [])) for r in analysis_results
                    )
                }
            )
            
        except Exception as e:
            self.logger.error(f"Analysis failed: {str(e)}", exc_info=True)
            return self._build_response(
                request=request,
                status="failed",
                errors=[{
                    "code": "ANALYSIS_ERROR",
                    "message": str(e),
                    "retryable": True
                }]
            )
    
    async def _load_chunks(self, project_id: str, sample_id: str) -> List[Dict]:
        """加载样本分块"""
        chunks_dir = Path(f"projects/{project_id}/samples/chunks/{sample_id}")
        
        chunk_files = sorted(chunks_dir.glob("chunk_*.json"))
        chunks = []
        
        for chunk_file in chunk_files:
            with open(chunk_file, 'r', encoding='utf-8') as f:
                chunk = json.load(f)
                chunks.append(chunk)
        
        return chunks
    
    async def _analyze_chunk(self, project_id: str, chunk: Dict, 
                           skills: Dict) -> Dict[str, Any]:
        """分析单个分块"""
        
        # 构建分析prompt
        prompt = self.prompt_builder.build_chunk_analysis_prompt(
            chunk_text=chunk["text"],
            chapter_range=chunk.get("chapter_range", ""),
            skills=skills
        )
        
        # 调用LLM
        response = await self.llm_client.generate(
            prompt=prompt,
            response_format="json"
        )
        
        # 记录指标
        self._record_llm_call(
            input_tokens=response["usage"]["prompt_tokens"],
            output_tokens=response["usage"]["completion_tokens"]
        )
        
        # 解析结果
        analysis = json.loads(response["content"])
        
        # 添加chunk元数据
        analysis["chunk_id"] = chunk["id"]
        analysis["chapter_range"] = chunk.get("chapter_range", "")
        
        return analysis
    
    async def _save_results(self, project_id: str, sample_id: str,
                           results: List[Dict]) -> List[str]:
        """保存分析结果"""
        output_dir = Path(f"projects/{project_id}/analysis/per_chunk/{sample_id}")
        output_dir.mkdir(parents=True, exist_ok=True)
        
        output_refs = []
        
        for result in results:
            chunk_id = result["chunk_id"]
            output_file = output_dir / f"{chunk_id}_analysis.json"
            
            with open(output_file, 'w', encoding='utf-8') as f:
                json.dump(result, f, ensure_ascii=False, indent=2)
            
            output_refs.append(str(output_file))
        
        return output_refs
```

```python
            # 1. 验证请求
            await self.validate_request(request)

            )
            
            # 3. 构建上下文包

            )
            
            # 7. 提取结构化信息
            structured_output = await self._extract_chapter_info(chapter_content)
            structured_output["chapter_number"] = chapter_number
            structured_output["word_count"] = len(chapter_content)

                request=request,
                status="success",

            )

        # 如果文件已存在，递增版本号
        while output_file.exists():
            version += 1
            output_file = output_dir / f"chapter_{chapter_number:03d}_v{version}.md"
        
        with open(output_file, 'w', encoding='utf-8') as f:
            f.write(content)
        
        self.logger.info(f"Draft saved: {output_file}")
        return str(output_file)
```

```python
import asyncio
from typing import Dict, Any, Optional
from litellm import acompletion
import json

class LLMClient:
    def __init__(self, model_config: Optional[Dict] = None):
        self.model_config = model_config or self._get_default_config()
    
    def _get_default_config(self) -> Dict:
        """获取默认模型配置"""
        return {
            "model": "gpt-4",
            "temperature": 0.7,
            "max_tokens": 4000
        }
    
    async def generate(self, prompt: str, 
                      response_format: Optional[str] = None,
                      **kwargs) -> Dict[str, Any]:
        """
        生成文本
        
        Args:
            prompt: 提示词
            response_format: 响应格式 (json/text)
            **kwargs: 额外参数
        
        Returns:
            {
                "content": str,
                "usage": {"prompt_tokens": int, "completion_tokens": int}
            }
        """
        messages = [{"role": "user", "content": prompt}]
        
        # 合并配置
        config = {**self.model_config, **kwargs}
        
        # 如果要求JSON格式
        if response_format == "json":
            config["response_format"] = {"type": "json_object"}
            # 在prompt中明确要求JSON
            messages[0]["content"] = f"{prompt}\n\n请以JSON格式返回结果。"
        
        try:
            response = await acompletion(
                model=config["model"],
                messages=messages,
                temperature=config.get("temperature", 0.7),
                max_tokens=config.get("max_tokens", 4000),
                **config.get("extra_params", {})
            )
            
            return {
                "content": response.choices[0].message.content,
                "usage": {
                    "prompt_tokens": response.usage.prompt_tokens,
                    "completion_tokens": response.usage.completion_tokens
                }
            }
            
        except Exception as e:
            raise Exception(f"LLM generation failed: {str(e)}")
    
    async def generate_with_retry(self, prompt: str, 
                                  max_retries: int = 3,
                                  **kwargs) -> Dict[str, Any]:
        """带重试的生成"""
        for attempt in range(max_retries):
            try:
                return await self.generate(prompt, **kwargs)
            except Exception as e:
                if attempt == max_retries - 1:
                    raise
                await asyncio.sleep(2 ** attempt)  # 指数退避
```

```python
class VectorStore:

            "text": "string",
            "vector": "vector(1536)",  # OpenAI embedding维度
            "type": "string",
            "metadata": "string"
        }
        
        if table_name not in db.table_names():
            db.create_table(table_name, schema=schema)

                         table_name: str = "knowledge_base"):

        table = db.open_table(table_name)
        
        # 批量添加
        table.add(documents)
    
    async def search(self, project_id: str, query_text: str = None,
                    query_vector: List[float] = None,
                    top_k: int = 10,
                    filters: Dict = None,
                    table_name: str = "knowledge_base") -> List[Dict]:
        """向量检索"""
        db = self._get_connection(project_id)
        table = db.open_table(table_name)
        
        if query_vector is None and query_text:
            # 生成query向量
            query_vector = await self._embed_text(query_text)
        
        # 执行检索
        results = table.search(query_vector).limit(top_k)
        
        # 应用过滤器
        if filters:
            for key, values in filters.items():
                if isinstance(values, list):
                    results = results.where(f"{key} IN {values}")
                else:
                    results = results.where(f"{key} = '{values}'")
        
        return results.to_list()
```

```python
# graph/graph_query.py
import networkx as nx
from pathlib import Path
import json
from typing import List, Dict, Any, Optional

class GraphQuery:
    def __init__(self):
        self.graphs = {}
    
    def _load_graph(self, project_id: str) -> nx.MultiDiGraph:
        """加载项目图谱"""
        if project_id not in self.graphs:
            graph_file = Path(f"projects/{project_id}/graph/graph.json")
            
            if graph_file.exists():
                with open(graph_file, 'r', encoding='utf-8') as f:
                    data = json.load(f)
                self.graphs[project_id] = nx.node_link_graph(
                    data, multigraph=True, directed=True
                )
            else:
                self.graphs[project_id] = nx.MultiDiGraph()
        
        return self.graphs[project_id]
    
    async def query_neighbors(self, project_id: str, node_id: str,
                             relation: Optional[str] = None,
                             hops: int = 1) -> List[Dict]:
        """查询邻居节点"""
        graph = self._load_graph(project_id)
        
        if node_id not in graph:
            return []
        
        # 使用ego_graph获取子图
        subgraph = nx.ego_graph(graph, node_id, radius=hops)
        
        results = []
        for node in subgraph.nodes():
            if node == node_id:
                continue
            
            node_data = graph.nodes[node]
            
            # 获取边信息
            edges = []
            for edge in graph.edges(node_id, node, data=True):
                edge_relation = edge[2].get('relation')
                if relation is None or edge_relation == relation:
                    edges.append({
                        'relation': edge_relation,
                        'attributes': edge[2]
                    })
            
            if edges or relation is None:
                results.append({
                    'node_id': node,
                    'node_type': node_data.get('type'),
                    'node_data': node_data,
                    'edges': edges
                })
        
        return results
    
    async def find_path(self, project_id: str, source: str,
                       target: str) -> List[str]:
        """查找路径"""
        graph = self._load_graph(project_id)
        
        try:
            return nx.shortest_path(graph.to_undirected(), source, target)
        except nx.NetworkXNoPath:
            return []
    
    async def query_by_type(self, project_id: str,
                          node_type: str) -> List[Dict]:
        """按类型查询节点"""
        graph = self._load_graph(project_id)
        
        results = []
        for node, data in graph.nodes(data=True):
            if data.get('type') == node_type:
                results.append({
                    'node_id': node,
                    'node_data': data
                })
        
        return results
```

```python
# text_processing/chapter_detector.py
import re
from typing import List, Dict, Tuple

class ChapterDetector:
    """章节识别器"""
    
    def __init__(self):
        # 中文章节标题模式
        self.patterns = [
            r'^第[零一二三四五六七八九十百千万\d]+章[：:\s]*.+$',
            r'^第[零一二三四五六七八九十百千万\d]+回[：:\s]*.+$',
            r'^Chapter\s+\d+[：:\s]*.+$',
            r'^#{1,3}\s*第[零一二三四五六七八九十百千万\d]+章',
        ]
        
        self.compiled_patterns = [re.compile(p, re.MULTILINE) for p in self.patterns]
    
    def detect_chapters(self, text: str) -> List[Dict]:
        """
        检测章节边界
        
        Returns:
            [
                {
                    "chapter_index": 1,
                    "title": "第一章 重生",
                    "start_offset": 0,
                    "end_offset": 5234
                },
                ...
            ]
        """
        chapters = []
        
        # 查找所有匹配的章节标题
        matches = []
        for pattern in self.compiled_patterns:
            for match in pattern.finditer(text):
                matches.append({
                    "title": match.group().strip(),
                    "start": match.start(),
                    "end": match.end()
                })
        
        # 按位置排序
        matches.sort(key=lambda x: x["start"])
        
        # 构建章节列表
        for i, match in enumerate(matches):
            chapter_index = i + 1
            title = match["title"]
            start_offset = match["start"]
            
            # 章节结束位置是下一章的开始位置
            if i < len(matches) - 1:
                end_offset = matches[i + 1]["start"]
            else:
                end_offset = len(text)
            
            chapters.append({
                "chapter_index": chapter_index,
                "title": title,
                "start_offset": start_offset,
                "end_offset": end_offset,
                "char_count": end_offset - start_offset
            })
        
        return chapters
    
    def extract_chapter_number(self, title: str) -> int:
        """从标题中提取章节号"""
        # 中文数字映射
        chinese_nums = {
            '零': 0, '一': 1, '二': 2, '三': 3, '四': 4,
            '五': 5, '六': 6, '七': 7, '八': 8, '九': 9,
            '十': 10, '百': 100, '千': 1000, '万': 10000
        }
        
        # 尝试提取阿拉伯数字
        match = re.search(r'\d+', title)
        if match:
            return int(match.group())
        
        # 尝试提取中文数字
        match = re.search(r'[零一二三四五六七八九十百千万]+', title)
        if match:
            return self._chinese_to_arabic(match.group(), chinese_nums)
        
        return 0
    
    def _chinese_to_arabic(self, chinese: str, mapping: Dict) -> int:
        """中文数字转阿拉伯数字"""
        result = 0
        temp = 0
        unit = 1
        
        for char in reversed(chinese):
            num = mapping.get(char, 0)
            
            if num >= 10:
                if num > unit:
                    unit = num
                else:
                    unit *= num
            else:
                temp += num * unit
        
        result += temp
        return result
```

```python
# text_processing/chunker.py
from typing import List, Dict

class Chunker:
    """文本分块器"""
    
    def __init__(self, max_chunk_size: int = 2500, overlap: int = 200):
        self.max_chunk_size = max_chunk_size
        self.overlap = overlap
    
    def chunk_by_chapters(self, text: str, chapters: List[Dict]) -> List[Dict]:
        """按章节分块"""
        chunks = []
        
        for chapter in chapters:
            chapter_text = text[chapter["start_offset"]:chapter["end_offset"]]
            
            # 如果章节长度小于最大块大小，直接作为一个块
            if len(chapter_text) <= self.max_chunk_size:
                chunks.append({
                    "id": f"chunk_{len(chunks):04d}",
                    "chapter_index": chapter["chapter_index"],
                    "part_index": 0,
                    "start_offset": chapter["start_offset"],
                    "end_offset": chapter["end_offset"],
                    "char_count": len(chapter_text),
                    "text": chapter_text,
                    "heading_path": chapter["title"]
                })
            else:
                # 章节过长，需要进一步分割
                sub_chunks = self._split_long_chapter(
                    chapter_text,
                    chapter["start_offset"],
                    chapter["chapter_index"],
                    chapter["title"]
                )
                chunks.extend(sub_chunks)
        
        return chunks
    
    def _split_long_chapter(self, text: str, base_offset: int,
                          chapter_index: int, chapter_title: str) -> List[Dict]:
        """分割长章节"""
        chunks = []
        part_index = 0
        start = 0
        
        while start < len(text):
            end = start + self.max_chunk_size
            
            # 如果不是最后一块，尝试在句号、问号、感叹号处断开
            if end < len(text):
                # 寻找最近的句子边界
                for i in range(end, max(start + self.max_chunk_size // 2, end - 200), -1):
                    if text[i] in '。！？\n':
                        end = i + 1
                        break
            
            chunk_text = text[start:end]
            
            chunks.append({
                "id": f"chunk_{len(chunks):04d}",
                "chapter_index": chapter_index,
                "part_index": part_index,
                "start_offset": base_offset + start,
                "end_offset": base_offset + end,
                "char_count": len(chunk_text),
                "text": chunk_text,
                "heading_path": f"{chapter_title} (Part {part_index + 1})"
            })
            
            # 下一块从overlap位置开始
            start = end - self.overlap if end < len(text) else end
            part_index += 1
        
        return chunks
    
    def chunk_without_chapters(self, text: str) -> List[Dict]:
        """无章节标题时按固定大小分块"""
        chunks = []
        start = 0
        synthetic_chapter_index = 1
        
        while start < len(text):
            end = min(start + self.max_chunk_size, len(text))
            
            chunk_text = text[start:end]
            
            chunks.append({
                "id": f"chunk_{len(chunks):04d}",
                "chapter_index": synthetic_chapter_index,
                "part_index": 0,
                "start_offset": start,
                "end_offset": end,
                "char_count": len(chunk_text),
                "text": chunk_text,
                "heading_path": f"Synthetic Chapter {synthetic_chapter_index}",
                "is_synthetic": True
            })
            
            start = end - self.overlap if end < len(text) else end
            synthetic_chapter_index += 1
        
        return chunks
```

### 6.3 覆盖率校验算法

```python
# text_processing/coverage_validator.py
from typing import List, Dict, Tuple

class CoverageValidator:
    """覆盖率校验器"""
    
    def validate(self, total_chars: int, chunks: List[Dict]) -> Dict:
        """
        验证分块覆盖率
        
        Returns:
            {
                "total_chars": int,
                "processed_chars": int,
                "coverage_ratio": float,
                "is_complete": bool,
                "missing_ranges": [(start, end), ...]
            }
        """
        # 排序chunks
        sorted_chunks = sorted(chunks, key=lambda x: x["start_offset"])
        
        # 合并重叠区间
        merged_ranges = self._merge_ranges([
            (chunk["start_offset"], chunk["end_offset"])
            for chunk in sorted_chunks
        ])
        
        # 计算处理字符数
        processed_chars = sum(end - start for start, end in merged_ranges)
        
        # 找出未覆盖区间
        missing_ranges = self._find_missing_ranges(merged_ranges, total_chars)
        
        coverage_ratio = processed_chars / total_chars if total_chars > 0 else 0
        
        return {
            "total_chars": total_chars,
            "processed_chars": processed_chars,
            "coverage_ratio": coverage_ratio,
            "is_complete": coverage_ratio >= 0.999,  # 允许0.1%误差
            "missing_ranges": missing_ranges
        }
    
    def _merge_ranges(self, ranges: List[Tuple[int, int]]) -> List[Tuple[int, int]]:
        """合并重叠区间"""
        if not ranges:
            return []
        
        ranges = sorted(ranges)
        merged = [ranges[0]]
        
        for start, end in ranges[1:]:
            last_start, last_end = merged[-1]
            
            if start <= last_end:
                # 有重叠，合并
                merged[-1] = (last_start, max(last_end, end))
            else:
                # 无重叠，添加新区间
                merged.append((start, end))
        
        return merged
    
    def _find_missing_ranges(self, covered_ranges: List[Tuple[int, int]],
                           total_chars: int) -> List[Tuple[int, int]]:
        """找出未覆盖区间"""
        missing = []
        expected_start = 0
        
        for start, end in covered_ranges:
            if start > expected_start:
                missing.append((expected_start, start))
            expected_start = max(expected_start, end)
        
        # 检查结尾
        if expected_start < total_chars:
            missing.append((expected_start, total_chars))
        
        return missing
```

### 6.4 伏笔债务计算算法

```python
# memory/foreshadowing_tracker.py
from typing import List, Dict
from datetime import datetime

class ForeshadowingTracker:
    """伏笔追踪器"""
    
    def calculate_debt_score(self, foreshadowing: Dict, 
                            current_chapter: int) -> float:
        """
        计算伏笔债务评分
        
        评分越高表示越需要推进或回收
        
        考虑因素：
        1. 埋设后经过的章节数
        2. 最后推进后经过的章节数
        3. 预期回收范围
        4. 伏笔类型（主线伏笔权重更高）
        """
        planted_chapter = foreshadowing["planted_chapter"]
        last_advanced_chapter = foreshadowing.get("last_advanced_chapter", planted_chapter)
        expected_payoff_start = foreshadowing.get("expected_payoff_range", {}).get("start")
        foreshadowing_type = foreshadowing.get("type", "normal")
        
        # 1. 埋设后经过的章节数
        chapters_since_planted = current_chapter - planted_chapter
        
        # 2. 最后推进后经过的章节数
        chapters_since_advanced = current_chapter - last_advanced_chapter
        
        # 3. 是否接近预期回收
        approaching_payoff = 0.0
        if expected_payoff_start:
            if current_chapter >= expected_payoff_start:
                approaching_payoff = 1.0
            elif current_chapter >= expected_payoff_start - 5:
                approaching_payoff = 0.5
        
        # 4. 类型权重
        type_weight = {
            "main": 1.5,      # 主线伏笔
            "subplot": 1.2,   # 支线伏笔
            "character": 1.0, # 人物伏笔
            "normal": 1.0
        }.get(foreshadowing_type, 1.0)
        
        # 计算债务评分
        debt_score = (
            (chapters_since_planted * 0.02) +      # 埋设时间
            (chapters_since_advanced * 0.05) +     # 未推进时间
            (approaching_payoff * 0.3)             # 接近回收期
        ) * type_weight
        
        # 归一化到0-1
        debt_score = min(debt_score, 1.0)
        
        return debt_score
    
    def get_high_debt_foreshadowing(self, foreshadowings: List[Dict],
                                   current_chapter: int,
                                   threshold: float = 0.6) -> List[Dict]:
        """获取高债务伏笔"""
        high_debt = []
        
        for fs in foreshadowings:
            if fs["status"] in ["planted", "advanced"]:
                debt_score = self.calculate_debt_score(fs, current_chapter)
                
                if debt_score >= threshold:
                    high_debt.append({
                        **fs,
                        "debt_score": debt_score
                    })
        
        # 按债务评分排序
        high_debt.sort(key=lambda x: x["debt_score"], reverse=True)
        
        return high_debt
```

## 10. 常见问题与解决方案

### 10.1 性能问题

**问题：大文件处理速度慢**
- 原因：一次性加载整个文件到内存
- 解决：使用流式读取，分块处理
- 优化：并行处理多个分块（控制并发数）

**问题：向量检索速度慢**
- 原因：向量库未建立索引
- 解决：创建IVF索引
- 优化：定期重建索引，调整分区数

**问题：LLM调用频繁超时**
- 原因：网络不稳定或模型负载高
- 解决：增加重试机制，使用指数退避
- 优化：使用批量API，减少调用次数

### 10.2 数据一致性问题

**问题：记忆更新与图谱不一致**
- 原因：更新逻辑分离，未保证原子性
- 解决：使用事务确保同时更新
- 优化：定期运行一致性检查工具

**问题：章节终稿后记忆未更新**
- 原因：记忆摄取Agent执行失败
- 解决：检查Agent日志，修复失败原因
- 优化：增加任务状态监控和告警

### 10.3 生成质量问题

**问题：章节越界，提前消耗后续章纲**
- 原因：边界控制未生效
- 解决：加强prompt中的约束，增加审查
- 优化：在生成时动态提醒剩余章纲

**问题：人设崩坏**
- 原因：记忆未正确加载或Canon缺失
- 解决：检查记忆文件和Canon规则
- 优化：增加人设一致性专项检查

**问题：AI味明显**
- 原因：去AI味Skill未生效
- 解决：强化去AI味规则，增加样本参考
- 优化：使用后处理工具进行二次润色
